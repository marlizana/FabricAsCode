BLOQUE 2 · TERRAFORM

Tu primer .tf completo, de principio a fin

main.tf

terraform {
  required_providers {
    fabric = {
      source  = "microsoft/fabric"
      version = "~> 1.14"
    }
  }
}
provider "fabric" {} # auth por variables de entorno
variable "capacity_id" { type = string }
resource "fabric_workspace" "bronze_dev" {
  display_name = "Bronze-DEV"
  capacity_id  = var.capacity_id
}
